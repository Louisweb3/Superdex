// ── Info box ──────────────────────────────────────────────────
function InfoBox({ children, type = 'info' }: { children: React.ReactNode; type?: 'info' | 'warning' | 'success' }) {
  const styles = {
    info:    { bg: 'hsl(var(--neon-blue)/0.06)',   border: 'hsl(var(--neon-blue)/0.25)',   icon: 'hsl(var(--neon-blue))'   },
    warning: { bg: 'hsl(var(--neon-orange)/0.06)', border: 'hsl(var(--neon-orange)/0.25)', icon: 'hsl(var(--neon-orange))' },
    success: { bg: 'hsl(var(--neon-green)/0.06)',  border: 'hsl(var(--neon-green)/0.25)',  icon: 'hsl(var(--neon-green))'  },
  };
  const s = styles[type];
  return (
    <div
      className="p-4 rounded-xl my-4 text-sm leading-relaxed text-[hsl(var(--muted-foreground))]"
      style={{ background: s.bg, border: `1px solid ${s.border}` }}
    >
      {children}
    </div>
  );
}
